                Object content = mail.getContent();
                if (content.getClass().isAssignableFrom(MimeMultipart.class)) {
                    MimeMultipart mimeMultipart = (MimeMultipart) content;
                    for (int i = 0; i < mimeMultipart.getCount(); i++) {

                             break
                         }
                    }
                }
